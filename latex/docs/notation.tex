\begin{notation}
  \begin{notationlist}{5.8em}
    \item[$N,\ M$] UAV 数量与 UE 数量
    \item[$H,\ \tau,\ T$] UAV 飞行高度、时隙时长与 episode 时隙数
    \item[$\mathbf q_u(t)$] UAV $u$ 在时隙 $t$ 的水平位置
    \item[$\mathbf w_m(t)$] 地面 UE $m$ 在时隙 $t$ 的水平位置
    \item[$\mathbf p_u(t),\ \mathbf p_m(t)$] UAV 与 UE 的三维位置
    \item[$v_{\max},\ d_{\min}$] UAV 最大速度与机间最小安全距离
    \item[$R_c,\ R_s$] UAV 物理服务覆盖半径与需求感知半径
    \item[$G,\ A,\ S$] 评估单元内实际生成、自然接纳、按时完成的服务请求数
    \item[$\mathrm{DSR}$] 端到端截止期满足率，$S/G$
    \item[$J,\ c_m$] Jain 公平性指数与 UE $m$ 的累计服务覆盖量
    \item[$g,\ R$] 链路信道功率增益与传输速率（bit/s）
    \item[$P^{\mathrm{comm}}$] 通信发射功率
    \item[$\sigma^2$] 噪声功率
    \item[$B^{\mathrm{edge}}$] UE-UAV 接入链路带宽
    \item[$B^{\mathrm{inter}}$] UAV 间链路带宽
    \item[$B^{\mathrm{backhaul}}$] UAV-MBS 回传链路带宽
    \item[$B_{\max},\ B_{\mathrm{low}}$] UE 电池容量与低电量阈值
    \item[$\beta$] 请求流行度 Zipf 指数
    \item[$\mathcal M_t,\ \mathcal J$] 时隙 $t$ 的自然接纳服务请求集合与执行节点集合
    \item[$\mathcal C_m$] 请求 $m$ 的合法执行节点集合
    \item[$\mathbf a,\ x_{mj}$] 请求分配向量与请求 $m$ 分配到节点 $j$ 的二元指示
    \item[$L_m,\ C_m$] 请求 $m$ 的输入字节数与所需 CPU 总周期
    \item[$\omega_m$] 请求 $m$ 每字节所需 CPU 周期数，$C_m=\omega_mL_m$
    \item[$B_s$] 服务程序 $s$ 的字节数
    \item[$F_u,\ F_{\mathrm{mbs}}$] UAV $u$ 与 MBS 的计算能力（周期/s）
    \item[$N_j,\ f_{mj}$] 分配到节点 $j$ 的最终请求数与请求分得的计算频率
    \item[$T_m,\ D_m$] 请求 $m$ 的完成时延与截止时间
    \item[$T^{\mathrm{tx}}_m$] 请求传输时延，$8L_m/R$
    \item[$T^{\mathrm{cmp}}_m,\ E^{\mathrm{cmp}}_m$] 请求计算时延与计算能耗
    \item[$\kappa$] 动态 CPU 功耗模型的有效芯片电容系数
    \item[$E_{\mathrm{assign}}$] 卸载分配相关的非飞行 UAV 能耗
    \item[$s_t,\ O_t$] 强化学习完整状态与上层 UAV 联合观测
    \item[$o_{u,t},\ a_{u,t}$] UAV $u$ 的局部预动作观测与连续原始动作提案
    \item[$a^{\mathrm{exec}}_{u,t}$] 环境映射后的 UAV 执行动作
    \item[$\pi_\theta,\ V_\phi$] 参数化策略与价值估计网络
    \item[$\theta,\ \phi$] 策略参数与价值网络参数
    \item[$r_t,\ \mathcal G_t$] 即时奖励与从时隙 $t$ 开始的折扣回报
    \item[$V_t^\pi,\ Q_t^\pi$] 状态值函数与动作值函数
    \item[$A_t^\pi,\ \widehat A_t$] 真实优势函数与优势估计
    \item[$\gamma$] 回报折扣因子
    \item[$\lambda_{\mathrm{GAE}}$] 广义优势估计的多步混合系数
    \item[$b_t,\ c_t$] 价值自举掩码与优势递推连接掩码
    \item[$\rho_t,\ \varepsilon$] PPO 新旧动作概率比率与裁剪系数
    \item[$Q,\ K,\ V$] 注意力查询、键与值矩阵（本体按上下文区分值函数）
    \item[$d_k$] 单个注意力头的键向量维度
    \item[$s_{mj},\ p_{mj}$] 给定配额下的缓存混合成功分数与公开缓存命中信念
    \item[$\delta_0,\ c_t^{\mathrm{sw}}$] 评分使用的名义软件费用与当次完整实测软件耗时
    \item[$\lambda,\ E_{\mathrm{norm}}$] 候选间能耗价格与开发阶段冻结的预测能耗归一化常数
  \end{notationlist}
\end{notation}
