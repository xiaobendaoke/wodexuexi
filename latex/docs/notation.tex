\begin{notation}

  \begin{notationlist}{2em}
    \item[$\displaystyle N$] UAV 数量
    \item[$\displaystyle M$] UE 数量
    \item[$\displaystyle H$] UAV 飞行高度
    \item[$\displaystyle \tau$] 时间步时长
    \item[$\displaystyle T$] episode 总时间步数
    \item[$\displaystyle R_c$] UAV 覆盖半径
    \item[$\displaystyle v_{max}$] UAV 最大飞行速度
    \item[$\displaystyle B_{max}$] UE 电池容量
    \item[$\displaystyle B_{low}$] UE 电池临界阈值
    \item[$\displaystyle P_{move}$] UAV 飞行功率
    \item[$\displaystyle P_{hover}$] UAV 悬停功率
    \item[$\displaystyle P_{wpt}$] WPT 发射功率
    \item[$\displaystyle f_{mbs}$] MBS 计算能力
    \item[$\displaystyle f_{uav}$] UAV 计算能力
    \item[$\displaystyle \eta$] WPT 能量采集效率
    \item[$\displaystyle \beta_0$] 参考距离 1m 时的信道功率增益
    \item[$\displaystyle DSR$] Deadline satisfaction rate
    \item[$\displaystyle P_{nat}$] 自然覆盖服务比例
    \item[$\displaystyle J$] Jain 公平性指数
    \item[$\displaystyle R_t$] 系统级综合奖励
    \item[$\displaystyle R_{lower}$] 下层局部训练奖励
    \item[$\displaystyle \lambda_{dsr}$] DSR Lagrange 乘子
    \item[$\displaystyle \lambda_{mbs}$] MBS load Lagrange 乘子
  \end{notationlist}

\end{notation}
