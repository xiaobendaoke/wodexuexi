\begin{notation}
  \begin{notationlist}{7.8em}
    \item[$\mathcal{U},\ N$] 无人机（UAV）集合与数量（主配置 $N=5$）
    \item[$M$] 地面用户设备（UE）总数量（主配置 $M=100$；第四章敏感性含 $60\sim 140$ 及 $3$）
    \item[$H,\ \tau,\ T$] UAV 固定飞行高度（$100\,\mathrm{m}$）、时隙长度（$1\,\mathrm{s}$）与回合步数（第三章 $T=1000$；第五章到达窗口 $200\,\mathrm{s}$）
    \item[$\mathbf{q}_u(t),\ \mathbf{w}_m(t)$] UAV $u$ 与地面 UE $m$ 在时隙 $t$ 的二维水平坐标（$\mathrm{m}$）
    \item[$\mathbf{p}_u(t),\ \mathbf{p}_m(t)$] UAV $u$ 与地面 UE $m$ 在时隙 $t$ 的三维空间坐标（$\mathrm{m}$）
    \item[$v_{\max},\ d_{\min}$] UAV 最大水平飞行速度（$15\,\mathrm{m/s}$）与机间最小水平安全间距（$200\,\mathrm{m}$）
    \item[$R_c,\ R_s$] UAV 自然覆盖服务半径（$100\,\mathrm{m}$）与需求/邻机感知半径（$460\,\mathrm{m}$）
    \item[$G,\ A,\ S$] 评估单元内实际生成、自然覆盖接纳、在截止期前完成的服务请求计数
    \item[$\mathrm{DSR},\ \alpha,\ \eta$] 端到端截止期满足率（$S/G$）、自然接纳率（$A/G$）与接纳后条件完成率（$S/A$）
    \item[$J,\ c_m(t)$] Jain 公平性指数与 UE $m$ 的累计历史达标服务覆盖频率
    \item[$g_{ij}(t),\ R,\ \sigma^2$] 三维无线链路功率增益、可达传输速率（$\mathrm{bit/s}$）与加性白高斯噪声功率（$10^{-13}\,\mathrm{W}$）
    \item[$P^{\mathrm{comm}},\ G_0g_0$] 无线发射功率（$0.5\,\mathrm{W}$）与单位距离自由空间信道参考增益常数（$3.244132\times 10^{-4}$）
    \item[$B^{\mathrm{edge}},\ B^{\mathrm{inter}},\ B^{\mathrm{bh}}$] 边缘接入带宽（$40\,\mathrm{MHz}$）、机间协作带宽（$20\,\mathrm{MHz}$）与回传带宽（$0.75\,\mathrm{MHz}$）
    \item[$\beta$] 服务与内容流行度 Zipf 分布偏斜指数（$\beta=0.8$）
    \item[$F_u,\ F_{\mathrm{M}}$] UAV $u$（$[75,60,70,85,50]\,\mathrm{GHz}$）与宏基站（$200\,\mathrm{GHz}$）的计算频率容量
    \item[$S_u^{\mathrm{cache}},\ B_s$] UAV $u$ 的机载缓存容量（$[158,171,148,134,162]\,\mathrm{MB}$）与文件 $s$ 的字节大小
    \item[$L_m,\ C_m,\ \omega_s$] 服务请求 $m$ 的输入数据字节数、所需总 CPU 周期数与每字节周期密度（$C_m=\omega_s L_m$）
    \item[$T_m,\ D_m$] 请求 $m$ 的端到端完成时延与截止时间预算（Deadline）
    \item[$U_m,\ I_{m,u,v},\ V_{j,s}$] 接入上传时间、机间协作输入转发时间与缓存未命中时的服务程序回源下载时间
    \item[$h_{j,s}(t),\ \widehat{h}_{v,s}$] 节点 $j$ 对文件 $s$ 的二值真实缓存命中指示与基于容量的邻机缓存命中概率信念
    \item[$\mathcal{M}(t),\ \mathcal{J}$] 时隙 $t$ 自然接纳的服务请求集合与执行节点集合
    \item[$b_m,\ a_m,\ \mathcal{C}_m$] 请求 $m$ 的目标执行节点分配变量与硬可行候选节点集合
    \item[$N_j(\mathbf{b}),\ f_{m,j}$] 单时隙协议中分配至 UAV $j$ 的最终批量请求数与等份共享计算频率 $F_j/N_j(\mathbf{b})$
    \item[$o_{u,t},\ \mathbf{o}_t$] 第三章上层单 UAV 定长局部观测（$295$ 维，含 $4$ 邻机槽与 $30$ 用户槽）与五机联合观测
    \item[$\mathbf{o}_u^{\mathrm{offload}},\ \mathbf{o}_u^{\mathrm{low}},\ \mathbf{r}_{u,m}$] 第三章下层单 UAV 卸载观测（$1405$ 维）、五维来源特征与百槽十四维请求特征
    \item[$\pi_\theta,\ V_\phi,\ V_\psi$] 参数化策略网络（Actor）与集中式状态价值网络（Critic）
    \item[$r_u,\ r_t^{\mathrm{low}}$] 第三章上层含违例惩罚的个体轨迹奖励与下层含自适应违例乘子的团队卸载奖励
    \item[$\lambda_{\mathrm{dsr}},\ \lambda_{\mathrm{mbs}}$] 第三章下层针对条件完成率目标与 MBS 负载上限的自适应惩罚乘子
    \item[$\gamma,\ \lambda_{\mathrm{GAE}},\ \varepsilon$] 强化学习折扣因子、广义优势估计（GAE）平滑系数与 PPO 裁剪系数
    \item[$\mathbf{Q}_u^h,\ \mathbf{K}_i^h,\ \mathbf{V}_i^h,\ d_h$] 多头交叉注意力第 $h$ 头的查询、键、值向量与单头维度（$d_h=8$）
    \item[{$F(\mathbf{a})=(F_1(\mathbf{a}),F_2(\mathbf{a}))$}] 第四章 MC-ITR 的严格二元字典序目标（超时请求总数 $F_1$ 与平均归一化时延 $F_2$）
    \item[$\mathcal{N}_1(\mathbf{a}),\ R_{\max}$] 第四章单请求迁移邻域集合与局部搜索最大迭代轮数预算
    \item[$\mathcal{K}(s),\ \mathcal{M}(s)$] 第五章动态排队协议下的候选分配方案集合（$K=4$）与候选五机联合位移集合（$M=6$）
    \item[$k_{\mathrm{safe}},\ R_{\mathrm{rescue}}(s,m)$] 第五章保守回退安全分配方案与待接纳紧急需求的空间救援评分
    \item[$\tilde{r}_{n,j},\ \tilde{\mathcal{C}}_j(\tau)$] 第五章释放时间串联排队近似中的任务 CPU 释放时刻与节点 $j$ 的就绪任务集合
    \item[$\widehat{C}(s,k,m),\ \widehat{O}(s,k,m)$] 释放时间串联排队近似输出的预计可按时完成量与预计排队溢出量
    \item[$\mathbf{c}(s),\ \mathbf{a}_k,\ \mathbf{m}_{km},\ \mathbf{x}_V(s)$] 第五章公共上下文（$14$ 维）、分配特征（$14$ 维）、运动特征（$16$ 维）与价值网络输入（$34$ 维）
    \item[{$\boldsymbol{\phi}_{km},\ \mathbf{C}_{\mathrm{upper}}(s,k)$}] 第五章运动核心交互四维特征向量与上层运动分支加权的期望反馈向量
    \item[$q_T(k,m\mid s),\ Q_{\mathrm{cf}}(s,k,m)$] 第五章三步规划教师软目标分布与共同随机数（CRN）三步反事实分支评分
    \item[$\Phi_{\mathrm{pre}}(s_t),\ \Phi_{\mathrm{post}}(s_t^+)$] 第五章动作前/后基于释放时间近似的任务完成与负载越界状态势函数
    \item[$\widetilde{p}(k,m),\ J(s,k,m)$] 第五章神经引导 Top-6 MPC 的熵校准提议先验与三步克隆前瞻综合评分
  \end{notationlist}
\end{notation}
